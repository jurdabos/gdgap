# Interactive
duckdb -readonly catalog/gdgap.ducklake

# Then get the relevant snapshot_id
SELECT max(snapshot_id) AS latest_snapshot_id FROM ducklake_snapshot;

# And now we can list the schemas using the snapshot_id BIGINT (update it in the example below)
SELECT schema_id, schema_name
FROM ducklake_schema
WHERE
    18 >= begin_snapshot AND
    (18 < end_snapshot OR end_snapshot IS NULL);

# And now we can list all the tables in a schema using the schema_id BIGINT
SELECT table_id, table_name
FROM ducklake_table
WHERE
    schema_id = 1 AND
    18 >= begin_snapshot AND
    (18 < end_snapshot OR end_snapshot IS NULL);

# For each given table, we can list the available top-level columns using the ducklake_column table:
SELECT column_id, column_name, column_type
FROM ducklake_column
WHERE
    table_id = 3 AND
    parent_column IS NULL AND
    18 >= begin_snapshot AND
    (18 < end_snapshot OR end_snapshot IS NULL)
ORDER BY column_order;

# Now list the available top-level columns from table with given id containing the string 'SEX':

# Now that we know the table structure we can query actual data from the Parquet files that store table data by joining data files with delete
SELECT data.path AS data_file_path, del.path AS delete_file_path
FROM ducklake_data_file AS data
LEFT JOIN (
    SELECT *
    FROM ducklake_delete_file
    WHERE
        18 >= begin_snapshot AND
        (18 < end_snapshot OR end_snapshot IS NULL)
    ) AS del
USING (data_file_id)
WHERE
    data.table_id = 2 AND
    18 >= data.begin_snapshot AND
    (18 < data.end_snapshot OR data.end_snapshot IS NULL)
ORDER BY file_order;

# Query the column size of sex-related info vs. the size of columns next to them for comparison (in our example, columns 7 and 72 hold them)
SELECT *
FROM ducklake_file_column_stats
WHERE table_id = 3
  AND column_id IN (6, 7, 8, 71, 72, 73)
ORDER BY column_id;
